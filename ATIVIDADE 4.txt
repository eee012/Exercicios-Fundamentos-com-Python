#3- Crie um programa que peça ao usuário para 
# digitar a sua altura e a altura de um amigo (em centímetros). 
# Verifique se a altura do usuário é maior que a do amigo e, se for, 
# exiba a mensagem "Você é mais alto que seu amigo".

print('Exercicio 3')

User_Height01 = float(input('Digite a sua altura: ' ))
User_Height02 = float(input('Digite a altura do seu amigo: ' ))

if User_Height01 == User_Height02:
    print()
    print('Voce tem a mesma altura do seu amigo!')
if User_Height01 > User_Height02:
    print()
    print('Voce e mais alto que seu amigo!')  
if User_Height01 < User_Height02:
    print()
    print('Voce e mais baixo que seu amigo!')

print()
print()
